import androidx.appcompat.app.AppCompatActivity;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;
import android.graphics.Color;
import android.os.Bundle;
import java.util.ArrayList;
import java.util.List;

public class MainActivity extends AppCompatActivity {
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        RecyclerView recyclerView = findViewById(R.id.recyclerView);
        recyclerView.setLayoutManager(new LinearLayoutManager(this));

        List<ColorItem> colorItems = new ArrayList<>();
        // Добавляем элементы с разными цветами
        colorItems.add(new ColorItem(Color.RED, Color.WHITE, "Красный"));
        colorItems.add(new ColorItem(Color.BLUE, Color.WHITE, "Синий"));
        colorItems.add(new ColorItem(Color.GREEN, Color.BLACK, "Зеленый"));
        colorItems.add(new ColorItem(Color.YELLOW, Color.BLACK, "Желтый"));
        colorItems.add(new ColorItem(Color.MAGENTA, Color.WHITE, "Пурпурный"));
        colorItems.add(new ColorItem(Color.CYAN, Color.BLACK, "Голубой"));
        colorItems.add(new ColorItem(Color.GRAY, Color.WHITE, "Серый"));
        colorItems.add(new ColorItem(Color.DKGRAY, Color.WHITE, "Темно-серый"));
        colorItems.add(new ColorItem(Color.parseColor("#FFA500"), Color.BLACK, "Оранжевый")); // Оранжевый
        colorItems.add(new ColorItem(Color.parseColor("#800080"), Color.WHITE, "Фиолетовый")); // Фиолетовый

        ColorAdapter adapter = new ColorAdapter(colorItems);
        recyclerView.setAdapter(adapter);
    }
}

public class ColorItem {
    private int backgroundColor;
    private int textColor;
    private String colorName;

    public ColorItem(int backgroundColor, int textColor, String colorName) {
        this.backgroundColor = backgroundColor;
        this.textColor = textColor;
        this.colorName = colorName;
    }

    public int getBackgroundColor() {
        return backgroundColor;
    }

    public int getTextColor() {
        return textColor;
    }

    public String getColorName() {
        return colorName;
    }
}

import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;
import java.util.List;

public class ColorAdapter extends RecyclerView.Adapter<ColorAdapter.ColorViewHolder> {
    private List<ColorItem> colorItems;

    public ColorAdapter(List<ColorItem> colorItems) {
        this.colorItems = colorItems;
    }

    @NonNull
    @Override
    public ColorViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_color, parent, false);
        return new ColorViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull ColorViewHolder holder, int position) {
        ColorItem item = colorItems.get(position);
        holder.colorName.setText(item.getColorName());
        holder.colorName.setBackgroundColor(item.getBackgroundColor());
        holder.colorName.setTextColor(item.getTextColor());
    }

    @Override
    public int getItemCount() {
        return colorItems.size();
    }

    static class ColorViewHolder extends RecyclerView.ViewHolder {
        TextView colorName;

        public ColorViewHolder(@NonNull View itemView) {
            super(itemView);
            colorName = itemView.findViewById(R.id.colorName);
        }
    }
}
